ที่ผ่านมา เวลาเราเปรียบเทียบว่าสองสิ่งเท่ากันหรือเปล่า เราใช้เครื่องหมายเท่ากับสามตัว แต่บางครั้ง เราอยากตรวจว่าสองสิ่งนั้นไม่เท่ากัน ซึ่งเราใช้เครื่องหมายตกใจตามด้วยเครื่องหมายเท่ากับสองตัว และโดยทั่วไปแล้ว ในภาษาโปรแกรม เครื่องหมายตกใจ หรือที่เพื่อนชาวอเมริกันของเราเรียกกันติดปากว่า เครื่องหมายตกใจ (_bang_) ก็หมายความว่า ไม่ ลองจินตนาการว่าคุณทำงานอยู่ในร้านค้าแห่งหนึ่ง แล้วฝ่ายบริหารตัดสินใจว่า ห้ามใส่ชุดดิสโก้เข้าร้านเด็ดขาด

เราพูดได้ว่า "ถ้าชุดไม่ใช่ชุดดิสโก้ ก็เปิดประตู" คำสั่งเงื่อนไข (_if statement_) ก็ทำงานแบบเดียวกันเป๊ะ

```javascript
if (outfit !== "disco") {
  openDoor()
}
```

แค่เงื่อนไขจะคืนค่าเป็นจริงเมื่อสองสิ่งนั้นต่างกัน และคืนค่าเป็นเท็จเมื่อสองสิ่งเหมือนกัน เราจึงเขียนเรื่องเดียวกันนี้ได้ว่า "ถ้าชุดเป็นชุดดิสโก้ ก็ไม่ต้องทำอะไร แต่ถ้าไม่ใช่ ก็เปิดประตู" แต่โดยทั่วไปแล้ว การใช้ตัวดำเนินการ (_operator_) `not` นั้นง่ายกว่าและเรียบร้อยกว่า เพื่อให้เรารู้และพูดในสิ่งที่ตัวเองหมายความได้จริงๆ

มีอีกเรื่องหนึ่งที่เกี่ยวข้อง บางครั้งคุณมีตัวแปรที่เป็นจริง แล้วอยากพลิกให้เป็นเท็จ หรือมีอะไรสักอย่างที่เป็นเท็จ แล้วอยากพลิกให้เป็นจริง เราใช้วิธีนี้เพื่อติดตามสถานะของบางสิ่งได้ ลองจินตนาการว่าเรามีอะไรสักอย่างเด้งไปมาจากซ้ายไปขวา และเรามีตัวแปรชื่อ `movingRight` (กำลังเคลื่อนที่ไปทางขวา) ซึ่งมีค่าเป็นจริงเมื่อถึงเวลาที่ต้องเคลื่อนที่ไปทางขวา แต่เมื่อชนขอบขวา แล้วอยากสลับ (_toggle_) ค่าให้เป็นเท็จ

เราทำแบบนี้ได้โดยพูดว่า "ถ้า `movingRight` เท่ากับจริง `movingRight` ก็เท่ากับเท็จ แต่ถ้าไม่ใช่ `movingRight` ก็เท่ากับจริง" นั่นก็เป็นวิธีหนึ่งในการสลับค่าของตัวแปร Jiki จะดูค่า `movingRight` ว่าเป็นจริงหรือเท็จ แล้วหาค่าที่ตรงกันข้าม และใส่ค่านั้นกลับเข้าไปในกล่อง

```javascript
if (movingRight === true) {
  movingRight = false
} else {
  movingRight = true
}
```

แต่มีวิธีที่เรียบร้อยกว่านี้ คือใช้เครื่องหมายตกใจตัวเดิมนี่แหละ

```javascript
movingRight = !movingRight
```

แค่นี้ก็พลิกค่าแล้ว `!movingRight` หมายถึงค่าที่ตรงกันข้ามกับค่าเดิมที่มีอยู่ก่อนหน้านี้

ดังนั้นถ้า `movingRight` เดิมเป็นจริง ตอนนี้ก็เป็นเท็จ ถ้าเดิมเป็นเท็จ ตอนนี้ก็เป็นจริง Jiki จะเอาค่าที่อยู่ในกล่องตอนนี้ แล้วเครื่องหมายตกใจก็บอก Jiki ให้หาค่าที่ตรงกันข้าม ค่าที่ตรงกันข้ามของจริงคือเท็จ ค่าที่ตรงกันข้ามของเท็จคือจริง แล้ว Jiki ก็ใส่ค่านั้นกลับเข้าไปในกล่อง แบบนี้ให้ผลเหมือนกับคำสั่ง if/else แบบยาวเมื่อกี้เป๊ะ แต่เขียนได้เรียบร้อยและกระชับกว่ามาก
